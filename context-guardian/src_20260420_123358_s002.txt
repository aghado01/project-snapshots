$j -lt $k; $j++)\n                {\n                    if ($closingHashes[$j] -eq $zeroHash) { continue }\n                    try\n                    {\n                        $d = Get-HammingDistance -Sig1 $closingHashes[$k] -Sig2 $closingHashes[$j]\n                        if ($d -lt $minH) { $minH = $d }\n                    }\n                    catch { Write-Verbose \"Hamming failed k=$k j=$j: $_\" }\n                }\n\n                if ($minH -ne [double]::MaxValue)\n                {\n                    $rec.MinHammingK = $minH\n                    $rec.ClosingRepeated = ($minH -lt $HammingThreshold)\n                }\n            }\n\n            # CTPH similarity vs previous turn\n            if ($k -gt 0 -and $ctphHashes[$k] -and $ctphHashes[$k - 1])\n            {\n                try\n                {\n                    $rec.CtphSimPrev = Compare-CTPHDirect -Hash1 $ctphHashes[$k] -Hash2 $ctphHashes[$k - 1]\n                }\n                catch { Write-Verbose \"CTPH compare failed k=$k: $_\" }\n            }\n        }\n\n        return $records.ToArray()\n    }\n}\n\nfunction Invoke-DecayAnalysis\n{\n    <#\n    .SYNOPSIS\n        Convenience wrapper: run decay analysis on a Copilot JSON/JSONL session.\n\n    .DESCRIPTION\n        Uses Get-MessagesFromRawSession from copilot-chat-recovery.psm1 to extract\n        turns, filters to assistant role, and passes them into Measure-DecaySignals.\n        Writes a CSV suitable for downstream Python/R/PowerShell analysis.\n\n    .PARAMETER SessionFile\n        Path to Copilot or Open Flaude session file (.json or .jsonl).\n\n    .PARAMETER AssistantRole\n        Role identifier for assistant messages. Defaults to <SessionFile>.decay.csv\n\n    .PARAMETER WindowSize\n        Rolling window K for MinHammingK.\n\n    .PARAMETER HammingThreshold\n        Threshold for ClosingRepeated flag.\n\n    .EXAMPLE\n        Invoke-DecayAnalysis -SessionFile './session.jsonl'\n\n    .EXAMPLE\n        Invoke-DecayAnalysis -SessionFile './session.jsonl' `\n            -AssistantRole 'GitHub Copilot' `\n            -OutputCsv './session.decay.csv'\n    #>\n    [CmdletBinding()]\n    param(\n        [Parameter(Mandatory)]\n        [string]$SessionFile,\n\n        [string]$AssistantRole = 'assistant',\n\n        [string]$OutputCsv,\n\n        [ValidateRange(1, 20)]\n        [int]$WindowSize = 3,\n\n        [ValidateRange(1, 32)]\n        [int]$HammingThreshold = 12\n    )\n\n    if (-not (Test-Path -LiteralPath $SessionFile -PathType Leaf))\n    {\n        throw \"Session file not found: $SessionFile\"\n    }\n\n    if (-not $OutputCsv)\n    {\n        $OutputCsv = \"$SessionFile.decay.csv\"\n    }\n\n    # Get messages via copilot-chat-recovery if available\n    if (-not (Get-Command Get-MessagesFromRawSession -ErrorAction SilentlyContinue))\n    {\n        throw \"Get-MessagesFromRawSession not found. Import copilot-chat-recovery.psm1 first.\"\n    }\n\n    Write-Verbose \"Reading messages from $SessionFile\"\n    $messages = Get-MessagesFromRawSession -SessionFile $SessionFile\n\n    Write-Verbose \"Computing decay signals (AssistantRole='$AssistantRole')\"\n    $features = $messages | Measure-DecaySignals -AssistantRole $AssistantRole -WindowSize $WindowSize -HammingThreshold $HammingThreshold\n\n    if ($features.Count -eq 0)\n    {\n        Write-Warning \"No assistant turns found for role '$AssistantRole'. Nothing to write.\"\n        return\n    }\n\n    Write-Verbose \"Writing CSV to $OutputCsv\"\n    $features |\n    Select-Object TurnIndex, AssistantTurnIndex, WordCount, ScaffoldFraction,\n    SimHashFull, SimHashClosing, CtphHash, RepetitionJaccard,\n    MinHammingK, CtphSimPrev, ClosingRepeated, ClosingSlice |\n    Export-Csv -LiteralPath $OutputCsv -NoTypeInformation\n\n    Write-Verbose \"Decay analysis written to $OutputCsv\"\n    Write-Output $OutputCsv\n}\n\nExport-ModuleMember -Function @(\n    'Measure-DecaySignals',\n    'Invoke-DecayAnalysis'\n) |
